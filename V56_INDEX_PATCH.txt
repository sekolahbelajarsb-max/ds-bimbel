D) JAVASCRIPT:
Sebelum </body> tambahkan:
<script>
(function(){
  const box=document.getElementById('publicJobs');
  if(!box)return;
  const escJob=s=>String(s??'').replace(/[&<>"']/g,m=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[m]));
  const wa=n=>String(n||'').replace(/\D/g,'').replace(/^0/,'62');
  function renderJobs(rows){
    if(!rows.length){
      box.innerHTML='<div class="job-empty">Saat ini belum ada lowongan kerja yang tersedia.</div>';
      return;
    }
    box.innerHTML=rows.map(x=>{
      const phone=wa(x.whatsapp);
      const msg=encodeURIComponent('Halo Ds Bimbel, saya tertarik melamar posisi '+(x.position||x.title||'')+'. Mohon informasi selanjutnya.');
      const apply=x.apply_url
        ? `<a class="btn primary" target="_blank" rel="noopener" href="${escJob(x.apply_url)}">Lamar Sekarang</a>`
        : (phone?`<a class="btn primary" target="_blank" rel="noopener" href="https://wa.me/${phone}?text=${msg}">Lamar via WhatsApp</a>`:'');
      const img=x.image_url||'';
      return `<article class="job-card">
        ${img?`<img src="${escJob(img)}" alt="${escJob(x.title||x.position||'Lowongan Kerja')}" loading="lazy">`:''}
        <div class="job-body">
          <span class="tag">LOWONGAN KERJA</span>
          <h3>${escJob(x.title||x.position)}</h3>
          <div class="job-meta">
            ${x.position?`<span class="job-pill">💼 ${escJob(x.position)}</span>`:''}
            ${x.location?`<span class="job-pill">📍 ${escJob(x.location)}</span>`:''}
            ${x.employment_type?`<span class="job-pill">🕐 ${escJob(x.employment_type)}</span>`:''}
            ${x.deadline?`<span class="job-pill">📅 s.d. ${escJob(String(x.deadline).slice(0,10))}</span>`:''}
          </div>
          ${x.qualification?`<p><b>Kualifikasi:</b><br>${escJob(x.qualification).replace(/\n/g,'<br>')}</p>`:''}
          ${x.description?`<p>${escJob(x.description).replace(/\n/g,'<br>')}</p>`:''}
          ${apply}
        </div>
      </article>`;
    }).join('');
  }
  fetch('/api/public/jobs')
    .then(r=>r.ok?r.json():Promise.reject(new Error('Gagal memuat lowongan')))
    .then(renderJobs)
    .catch(()=>{box.innerHTML='<div class="job-empty">Lowongan kerja belum dapat dimuat. Silakan coba beberapa saat lagi.</div>'});
})();
</script>
