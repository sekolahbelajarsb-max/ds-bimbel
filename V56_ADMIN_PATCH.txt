C) JAVASCRIPT:
Tambahkan fungsi berikut di area script admin:

function resetJobForm(){
  ['jobEditId','jobTitle','jobPosition','jobImage','jobLocation','jobDeadline','jobQualification','jobDescription','jobRequirements','jobWa','jobApplyUrl'].forEach(id=>{const e=document.getElementById(id);if(e)e.value=''});
  if(document.getElementById('jobType'))jobType.value='Part Time';
  if(document.getElementById('jobStatus'))jobStatus.value='Aktif';
  if(document.getElementById('jobOrder'))jobOrder.value='0';
}

async function loadJobs(){
  try{
    const rows=await api('/jobs');
    window.jobsCache=rows||[];
    renderJobsAdmin();
  }catch(e){toast('Gagal memuat lowongan: '+e.message)}
}

function renderJobsAdmin(){
  const rows=document.getElementById('jobRows');
  if(!rows)return;
  const a=window.jobsCache||[];
  rows.innerHTML=a.length?a.map(x=>`<tr>
    <td>${x.image_url?`<img src="${esc(x.image_url)}" style="width:90px;height:55px;object-fit:cover;border-radius:8px">`:'-'}</td>
    <td><b>${esc(x.title)}</b></td>
    <td>${esc(x.position)}</td>
    <td>${esc(x.location||'-')}</td>
    <td>${esc(x.employment_type||'-')}</td>
    <td>${esc(x.deadline?String(x.deadline).slice(0,10):'-')}</td>
    <td><span class="badge">${esc(x.status)}</span></td>
    <td><button class="btn light" onclick="editJob(${JSON.stringify(x.id)})">Edit</button>
    <button class="btn danger" onclick="deleteJob(${JSON.stringify(x.id)})">Hapus</button></td>
  </tr>`).join(''):'<tr><td colspan="8" class="empty">Belum ada lowongan.</td></tr>';
}

function editJob(id){
  const x=(window.jobsCache||[]).find(a=>String(a.id)===String(id));
  if(!x)return;
  jobEditId.value=x.id;jobTitle.value=x.title||'';jobPosition.value=x.position||'';jobImage.value=x.image_url||'';
  jobLocation.value=x.location||'';jobType.value=x.employment_type||'Part Time';jobDeadline.value=x.deadline?String(x.deadline).slice(0,10):'';
  jobQualification.value=x.qualification||'';jobDescription.value=x.description||'';jobRequirements.value=x.requirements||'';
  jobWa.value=x.whatsapp||'';jobApplyUrl.value=x.apply_url||'';jobStatus.value=x.status||'Aktif';jobOrder.value=x.sort_order||0;
  document.getElementById('jobFormPanel').classList.remove('hidden');
}

async function saveJob(){
  if(!jobTitle.value.trim()||!jobPosition.value.trim()){toast('Judul dan posisi wajib diisi.');return}
  const data={
    title:jobTitle.value.trim(),position:jobPosition.value.trim(),image_url:jobImage.value.trim(),
    location:jobLocation.value.trim(),employment_type:jobType.value,deadline:jobDeadline.value||null,
    qualification:jobQualification.value.trim(),description:jobDescription.value.trim(),requirements:jobRequirements.value.trim(),
    whatsapp:jobWa.value.trim()||'0899-4515-669',apply_url:jobApplyUrl.value.trim(),status:jobStatus.value,
    sort_order:Number(jobOrder.value||0)
  };
  try{
    const id=jobEditId.value;
    await api(id?'/jobs/'+encodeURIComponent(id):'/jobs',{
      method:id?'PATCH':'POST',
      body:JSON.stringify(data)
    });
    toast(id?'Lowongan diperbarui.':'Lowongan berhasil ditambahkan.');
    resetJobForm();
    document.getElementById('jobFormPanel').classList.add('hidden');
    await loadJobs();
  }catch(e){toast('Gagal menyimpan: '+e.message)}
}

async function deleteJob(id){
  if(!confirm('Hapus lowongan ini?'))return;
  try{await api('/jobs/'+encodeURIComponent(id),{method:'DELETE'});toast('Lowongan dihapus.');await loadJobs()}
  catch(e){toast('Gagal menghapus: '+e.message)}
}
